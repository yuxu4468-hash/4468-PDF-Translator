## Goal

让译文里的**数字**与原文一致：翻译后逐条校验，失配的条目**定向重试一次**，
仍失配则**回退原文并告警**，且失配明细**只报一次**。

同时产出**真实失配率**这个数据——它决定阶段 2、3 的优先级。

## Architecture

校验发生在**翻译批次内部**、`_assemble()` 拼装成段落之前。

选择这一层的原因（分层依据）：`Item` 是真正的翻译单元（长段落会被
`split_long_text` 切成多个 chunk），每个 chunk 独立翻译，因此数值比对必须在
**item 级**才能对得上；而"回退原文"是用户可见行为，必须作用在**paragraph 级**。
两级各司其职，`_assemble()` 正好是交界处——它已经实现了"没有译文就置空 target
并记 error"，回退语义天然复用现成路径。

```
批次返回 → _apply() 落盘 item.result
            ↓
         _validate_numbers(items)          ← 新增
            ├ 全部通过 → 走原有流程
            └ 有失配   → 追加数字约束，单独重试这一小批   ← 新增
                          ↓
                     再校验
                        ├ 通过 → 采用
                        └ 失配 → item.result = ""（回退原文）+ 去重告警
                                   ↓
                              _assemble() 置 paragraph.target = ""
                                   ↓
                              排版阶段跳过（既不擦除也不绘制，原段保持德文）
```

```python
# ---------------------------------------------------------------------------
# 数值抽取与比对（用于译文的数字保护）
# ---------------------------------------------------------------------------

_NUMBER_RE = re.compile(r"\d+(?:[.,]\d+)*")

#: 汉字数字
_CN_DIGITS = {"〇": 0, "零": 0, "一": 1, "二": 2, "两": 2, "三": 3, "四": 4,
              "五": 5, "六": 6, "七": 7, "八": 8, "九": 9}
_CN_UNITS = {"十": 10, "百": 100, "千": 1000}
_CN_BIG = {"万": 10 ** 4, "亿": 10 ** 8}
_CN_NUM_CHARS = set(_CN_DIGITS) | set(_CN_UNITS) | set(_CN_BIG)

#: 允许跟在汉字数词后面的量词。**刻意取窄**：只有这些才认为数词独立成词。
#: 加进"部""分"会把「一部分」转成 1，凭空造出原文没有的数字。
_CN_TRAILERS = set("条类款项级种章节次组号年月日周")


def _is_cjk_char(char: str) -> bool:
    return "\u4e00" <= char <= "\u9fff"


def _number_candidates(token: str) -> set[str]:
    """一个数字串可能的归一化形式。

    千分位与小数点存在真实歧义：德语 `1.000` 是「一千」，英语 `1.000` 是「一点零」。
    无法只凭字符串判定，所以**两种读法都产出**，比对时命中任一即算通过 ——
    这是为了压住误报（宁可放过，不可误杀）。
    """
    token = (token or "").strip()
    if not token:
        return set()

    if "," in token and "." in token:
        # 后出现的那个是小数点，另一个是千分位
        if token.rfind(",") > token.rfind("."):
            variants = {token.replace(".", "").replace(",", ".")}
        else:
            variants = {token.replace(",", "")}
    else:
        sep = "," if "," in token else ("." if "." in token else "")
        if not sep:
            variants = {token}
        else:
            parts = token.split(sep)
            variants = {parts[0] + "." + "".join(parts[1:])}
            if all(len(part) == 3 for part in parts[1:]):
                variants.add("".join(parts))

    cleaned: set[str] = set()
    for item in variants:
        if "." in item:
            head, _, tail = item.partition(".")
            cleaned.add((head.lstrip("0") or "0") + "." + tail)
        else:
            cleaned.add(item.lstrip("0") or "0")
    return cleaned


def _chinese_to_int(run: str) -> int | None:
    """把「二十五」「两百」「二〇二〇」这类汉字数词转成整数；无法解析返回 None。"""
    if not run:
        return None
    total = 0
    section = 0
    number = 0
    for char in run:
        if char in _CN_DIGITS:
            number = _CN_DIGITS[char]
        elif char in _CN_UNITS:
            section += (number or 1) * _CN_UNITS[char]
            number = 0
        elif char in _CN_BIG:
            section = (section + number) * _CN_BIG[char]
            total += section
            section = 0
            number = 0
        else:
            return None
    return total + section + number


def chinese_numerals(text: str) -> list[int]:
    """抽取**独立成词**的汉字数词。

    保守是刻意的。若在汉字词组内部乱转（`一般`→1、`十分`→10），会凭空造出
    原文没有的数字，反而可能掩盖真正的漏译。所以数词串必须满足：

    - 前一个字符不是汉字，**或者是「第」**（`第五条`）；
    - 后一个字符不是汉字，**或者属于量词表**（`二十三种`）。
    """
    values: list[int] = []
    index = 0
    length = len(text)
    while index < length:
        if text[index] not in _CN_NUM_CHARS:
            index += 1
            continue
        start = index
        while index < length and text[index] in _CN_NUM_CHARS:
            index += 1
        run = text[start:index]
        prev_char = text[start - 1] if start > 0 else ""
        next_char = text[index] if index < length else ""
        prev_ok = (not prev_char) or (not _is_cjk_char(prev_char)) or prev_char == "第"
        next_ok = (not next_char) or (not _is_cjk_char(next_char)) or next_char in _CN_TRAILERS
        if prev_ok and next_ok:
            value = _chinese_to_int(run)
            if value is not None:
                values.append(value)
    return values


def extract_numbers(text: str) -> set[str]:
    """抽取文本里所有数值（阿拉伯数字 + 独立成词的汉字数词），返回归一化形式集合。"""
    if not text:
        return set()
    found: set[str] = set()
    for token in _NUMBER_RE.findall(text):
        found |= _number_candidates(token)
    for value in chinese_numerals(text):
        found.add(str(value))
    return found


def missing_numbers(source: str, target: str) -> list[str]:
    """返回**源里出现、译文里找不到**的数值。

    **单向包含**：不检查译文多出来的数值。译文里的数字比原文多通常是正常的
    —— `1. Juni 2018` 正确译成中文是 `2018年6月1日`，月份名词转成了数字 `6`。
    做对称比较会把这份正确译文误杀。
    """
    if not source:
        return []
    target_numbers = extract_numbers(target)
    if not target_numbers:
        target_numbers = set()

    missing: list[str] = []
    seen: set[str] = set()
    for token in _NUMBER_RE.findall(source):
        candidates = _number_candidates(token)
        if not candidates or (candidates & target_numbers):
            continue
        if token not in seen:
            seen.add(token)
            missing.append(token)
    for value in chinese_numerals(source):
        key = str(value)
        if key not in target_numbers and key not in seen:
            seen.add(key)
            missing.append(key)
    return missing
```

```python
    # ------------------------------------------------------------------
    def _validate_numbers(self, items: list[Item]) -> dict[int, list[str]]:
        """校验条目译文里的数字，返回 {item.key: 缺失的数值}。

        只对**没有 error 且已有译文**的条目做校验。单向包含判定见
        `text_utils.missing_numbers` 的文档。
        """
        problems: dict[int, list[str]] = {}
        for item in items:
            if item.error or not item.result:
                continue
            missing = missing_numbers(item.text, item.result)
            if missing:
                problems[item.key] = missing
        return problems

    def _report_numeric(self, source: str, missing: list[str]) -> None:
        """输出失配明细，按 (原文片段, 缺失数值) 去重，且每文档不超过上限。

        批量跑几十份资料时，同一个失配模式可能触发几百次；逐段刷屏会让日志
        失去可读性，而统计数字仍然完整保留，不影响判断。
        """
        try:
            limit = int(self.config.get("numeric_log_limit", 5))
        except (TypeError, ValueError):
            limit = 5
        if limit <= 0:
            return
        key = (source.strip()[:40], tuple(missing))
        with self._lock:
            if key in self._numeric_reported:
                return
            if len(self._numeric_reported) >= limit:
                return
            self._numeric_reported.add(key)
        self._log(
            "数值校验未通过：%r 缺 %s"
            % (source.strip()[:40], "、".join(missing[:6]))
        )
```

        # ---- 数值校验：失配则定向重试一次，仍失配则回退原文 ----
        if bool(self.config.get("numeric_guard", True)):

### Task 1.6 — 新增回归测试

**Files**
- **新建**：`tests/test_numeric_guard.py`
- 修改：`tests/run_all.py`

**Steps**

**1.6.1** 创建 `tests/test_numeric_guard.py`：

```python
"""数值保护回归测试。

锁住三件事：
  1. 正确译文不被误杀（单向包含 + 归一化）；
  2. 真实漏译能被抓到；
  3. 失配后回退原文、且不静默通过。

用法：
    python tests/test_numeric_guard.py
"""

from __future__ import annotations

import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))

from src.core.translator import TranslateStats  # noqa: E402
from src.shared.constants import DEFAULT_CONFIG  # noqa: E402
from src.shared.text_utils import (  # noqa: E402
    chinese_numerals,
    extract_numbers,
    missing_numbers,
)

FAILURES: list[str] = []


def check(condition, label):
    print(f"  [{'PASS' if condition else 'FAIL'}] {label}")
    if not condition:
        FAILURES.append(label)


def test_normalization():
    print("\n[1] 归一化：写法差异不算失配")
    check(missing_numbers("1,5 m", "1.5 米") == [], "德语小数逗号 1,5 ↔ 1.5")
    check(missing_numbers("5 x 1 m", "5×1 米") == [], "乘号写法 5 x 1 ↔ 5×1")
    check(missing_numbers("22-28 °C", "22–28 ℃") == [], "区间连字符 - ↔ –")
    check(missing_numbers("1.000 m", "1000 米") == [], "千分位 1.000 ↔ 1000")
    check(missing_numbers("1.000 m", "1.0 米") == [], "1.000 也允许按小数读作 1.0")


def test_asymmetric():
    print("\n[2] 单向包含：译文多出的数字不算失配")
    check(missing_numbers("Essen, 1. Juni 2018", "埃森，2018年6月1日") == [],
          "月份名词转数字（Juni → 6）不得误杀")
    extra = missing_numbers("Temperaturen 22 bis 28", "温度 22 至 28 摄氏度")
    check(extra == [], f"正常译文无失配（{extra}）")


def test_real_miss():
    print("\n[3] 真实漏译必须抓到")
    miss = missing_numbers("Essen, 1. Juni 2018", "埃森，2018 年")
    check(bool(miss), f"丢失的日与月被检出（{miss}）")
    check("1" in miss and "6" in miss, f"缺的正是 1 与 6（得到 {miss}）")

    miss2 = missing_numbers("Gesamtlaenge 50-60 cm", "全长 50 厘米")
    check("60" in miss2, f"区间上端丢失被检出（{miss2}）")


def test_chinese_numerals():
    print("\n[4] 中文数字参与比对")
    check(missing_numbers("5 Stueck", "第五条") == [], "第五条 ↔ 5")
    check(missing_numbers("23 Arten", "二十三种") == [], "二十三种 ↔ 23")
    check(missing_numbers("2020", "二〇二〇年") == [], "二〇二〇年 ↔ 2020")
    check(missing_numbers("1 Teil", "一般") != [], "「一般」不得被当成 1")
    check(missing_numbers("10 Grad", "十分") != [], "「十分」不得被当成 10")

    values = chinese_numerals("第五条 二十三种 一般 十分 二〇二〇年")
    check(5 in values and 23 in values and 2020 in values,
          f"独立成词的数词被正确解析（{values}）")
    check(1 not in values and 10 not in values,
          "词组内部的数词未被误解析")


def test_fingerprint():
    print("\n[5] 数值集合抽取")
    got = extract_numbers("要 50-60 cm，湿度 60 %，温度 22 °C")
    check({"50", "60", "22"} <= got, f"阿拉伯数字被抽出（{sorted(got)}）")


def test_stats_field():
    print("\n[6] 统计字段与开关")
    data = TranslateStats().to_dict()
    check("numeric_rejected" in data and "numeric_retries" in data,
          "TranslateStats 暴露数值校验统计")
    check(DEFAULT_CONFIG.get("numeric_guard") is True, "numeric_guard 默认开启")
    check(DEFAULT_CONFIG.get("numeric_log_limit") == 5, "明细上限默认 5 条")


def main() -> int:
    print("=" * 72)
    print("数值保护回归测试")
    print("=" * 72)
    test_normalization()
    test_asymmetric()
    test_real_miss()
    test_chinese_numerals()
    test_fingerprint()
    test_stats_field()
    print()
    print("=" * 72)
    if FAILURES:
        print(f"结果：{len(FAILURES)} 项未通过")
        for item in FAILURES:
            print(f"  - {item}")
        return 1
    print("数值保护测试全部通过 ✓")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
"""在一份真实文档上测数值失配率（走真实翻译，用缓存则零 token）。"""

from __future__ import annotations

import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent.parent
sys.path.insert(0, str(ROOT))

from src.core import pdf_parser, pipeline  # noqa: E402
from src.core.task import Task  # noqa: E402
from src.shared.config_loader import load_config  # noqa: E402
from src.shared.path_helpers import ensure_dirs  # noqa: E402

SRC = Path(r"<语料目录>"
           r"\01_官方准则_原文\DGHT_无尾目(蛙类)饲养准则_德文.pdf")


def main() -> int:
    ensure_dirs()
    config = load_config()
    config.update({"mode": "mono", "target_lang": "zh", "export_pdf": False,
                   "export_markdown": False, "page_range": "all", "ocr_mode": "off"})

    document = pdf_parser.parse_pdf(str(SRC), "m", SRC.name)
    task = Task(task_id="measure-numeric", file_id="m", filename=SRC.name)
    pipeline.run_task(task, document, config)

    stats = task.stats or {}
    total = stats.get("translated", 0)
    rejected = stats.get("numeric_rejected", 0)
    retries = stats.get("numeric_retries", 0)
    print("文档：%s" % SRC.name)
    print("  已翻译段数     %d" % total)
    print("  数值失配回退段 %d" % rejected)
    print("  定向重试次数   %d" % retries)
    if total:
        print("  失配率         %.2f%%" % (100.0 * rejected / total))
    document.close()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
